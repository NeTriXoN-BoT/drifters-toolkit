# Fixtures with their own core object must use the same archive/backup
# implementation as production; isolated parser/save tests remain independent.
get_property(DRIFTER_ALL_TARGETS DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target IN LISTS DRIFTER_ALL_TARGETS)
  get_target_property(DRIFTER_TARGET_SOURCES ${target} SOURCES)
  if("${DRIFTER_TARGET_SOURCES}" MATCHES "(^|;)src/core\\.cpp(;|$)")
    target_sources(${target} PRIVATE src/library_backup.cpp)
    target_link_libraries(${target} PUBLIC drifter_archive)
  endif()
endforeach()
